## 1. 自动备份脚本

### backup-database.sh - 数据库自动备份

```bash
#!/bin/bash
# ============================================================
# 数据库自动备份脚本
# 功能: 自动备份MariaDB数据库并清理旧备份
# 作者: 哈雷酱 (￣▽￣)／
# ============================================================

set -e  # 遇到错误立即退出

# ========== 配置区域 ==========
PROJECT_DIR="/home/monika/my-LNMP-project"
BACKUP_DIR="${PROJECT_DIR}/backups"
CONTAINER_NAME="my_lnmp_db"
DB_ROOT_PASSWORD="liu20041016"  # ⚠️ 建议从.env读取
RETENTION_DAYS=7  # 保留最近7天的备份
LOG_FILE="${PROJECT_DIR}/logs/backup.log"

# ========== 函数定义 ==========
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "${LOG_FILE}"
}

error_exit() {
    log "❌ 错误: $1"
    exit 1
}

# ========== 主逻辑 ==========
log "🚀 开始数据库备份任务..."

# 1. 确保备份目录存在
mkdir -p "${BACKUP_DIR}" || error_exit "无法创建备份目录"

# 2. 生成备份文件名
TIMESTAMP=$(date +%Y%m%d-%H%M%S)
BACKUP_FILE="${BACKUP_DIR}/backup-all-${TIMESTAMP}.sql.gz"

# 3. 检查容器是否运行
if ! docker ps | grep -q "${CONTAINER_NAME}"; then
    error_exit "数据库容器未运行!"
fi

# 4. 执行备份
log "📦 备份数据库到: ${BACKUP_FILE}"
if docker exec "${CONTAINER_NAME}" mysqldump \
    -uroot \
    -p"${DB_ROOT_PASSWORD}" \
    --all-databases \
    --single-transaction \
    --quick \
    --lock-tables=false \
    | gzip > "${BACKUP_FILE}"; then

    # 检查备份文件大小
    BACKUP_SIZE=$(du -h "${BACKUP_FILE}" | cut -f1)
    log "✅ 备份成功! 文件大小: ${BACKUP_SIZE}"
else
    error_exit "备份失败!"
fi

# 5. 清理旧备份
log "🗑️  清理${RETENTION_DAYS}天前的备份..."
DELETED=$(find "${BACKUP_DIR}" -name "backup-*.sql.gz" -mtime +${RETENTION_DAYS} -delete -print | wc -l)
log "✅ 已删除 ${DELETED} 个旧备份文件"

# 6. 显示备份列表
log "📋 当前备份列表:"
ls -lh "${BACKUP_DIR}" | grep "backup-" | tee -a "${LOG_FILE}"

log "🎉 备份任务完成!"
```

### 使用方法:
```bash
# 1. 创建脚本
cd /home/monika/my-LNMP-project/
vim backup-database.sh
# (粘贴上面的脚本内容)

# 2. 添加执行权限
chmod +x backup-database.sh

# 3. 手动执行测试
./backup-database.sh

# 4. 添加到crontab(每天凌晨2点自动备份)
crontab -e
# 添加这一行:
0 2 * * * /home/monika/my-LNMP-project/backup-database.sh

# 5. 查看备份日志
tail -f logs/backup.log
```

## 2. 健康检查脚本

### health-check.sh - 全面健康检查

```bash
#!/bin/bash
# ============================================================
# Docker LNMP 健康检查脚本
# 功能: 检查所有服务健康状态并发送告警
# 作者: 哈雷酱 (￣▽￣)／
# ============================================================

set -e

# ========== 配置 ==========
PROJECT_NAME="my_lnmp"
NGINX_CONTAINER="${PROJECT_NAME}_nginx"
PHP_CONTAINER="${PROJECT_NAME}_php"
DB_CONTAINER="${PROJECT_NAME}_db"
DB_ROOT_PASSWORD="liu20041016"

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
NC='\033[0m' # No Color

# ========== 检查函数 ==========
check_container_running() {
    local container=$1
    if docker ps | grep -q "${container}"; then
        echo -e "${GREEN}✅${NC} 容器 ${container} 运行中"
        return 0
    else
        echo -e "${RED}❌${NC} 容器 ${container} 未运行!"
        return 1
    fi
}

check_nginx() {
    echo "🔍 检查Nginx服务..."

    # 检查容器运行状态
    check_container_running "${NGINX_CONTAINER}" || return 1

    # 检查配置文件
    if docker exec "${NGINX_CONTAINER}" nginx -t 2>&1 | grep -q "successful"; then
        echo -e "${GREEN}✅${NC} Nginx配置正确"
    else
        echo -e "${RED}❌${NC} Nginx配置错误!"
        return 1
    fi

    # 检查HTTP响应
    if curl -f -s -o /dev/null http://localhost; then
        echo -e "${GREEN}✅${NC} Nginx HTTP响应正常"
    else
        echo -e "${YELLOW}⚠️${NC}  Nginx HTTP响应异常"
        return 1
    fi

    return 0
}

check_php() {
    echo "🔍 检查PHP-FPM服务..."

    # 检查容器运行状态
    check_container_running "${PHP_CONTAINER}" || return 1

    # 检查PHP版本
    PHP_VERSION=$(docker exec "${PHP_CONTAINER}" php -v 2>&1 | head -n1)
    echo -e "${GREEN}✅${NC} ${PHP_VERSION}"

    # 检查PHP-FPM进程
    if docker exec "${PHP_CONTAINER}" pgrep php-fpm >/dev/null 2>&1; then
        echo -e "${GREEN}✅${NC} PHP-FPM进程正常"
    else
        echo -e "${RED}❌${NC} PHP-FPM进程异常!"
        return 1
    fi

    return 0
}

check_database() {
    echo "🔍 检查数据库服务..."

    # 检查容器运行状态
    check_container_running "${DB_CONTAINER}" || return 1

    # 检查数据库连接
    if DB_VERSION=$(docker exec "${DB_CONTAINER}" mysql -uroot -p"${DB_ROOT_PASSWORD}" -e "SELECT VERSION();" 2>/dev/null | tail -n1); then
        echo -e "${GREEN}✅${NC} 数据库连接正常 (${DB_VERSION})"
    else
        echo -e "${RED}❌${NC} 数据库连接失败!"
        return 1
    fi

    # 检查数据库列表
    DB_COUNT=$(docker exec "${DB_CONTAINER}" mysql -uroot -p"${DB_ROOT_PASSWORD}" -e "SHOW DATABASES;" 2>/dev/null | wc -l)
    echo -e "${GREEN}✅${NC} 数据库数量: $((DB_COUNT - 1))"

    return 0
}

check_disk_space() {
    echo "🔍 检查磁盘空间..."

    DISK_USAGE=$(df -h / | tail -n1 | awk '{print $5}' | sed 's/%//')

    if [ "${DISK_USAGE}" -lt 80 ]; then
        echo -e "${GREEN}✅${NC} 磁盘使用率: ${DISK_USAGE}%"
    elif [ "${DISK_USAGE}" -lt 90 ]; then
        echo -e "${YELLOW}⚠️${NC}  磁盘使用率: ${DISK_USAGE}% (建议清理)"
    else
        echo -e "${RED}❌${NC} 磁盘使用率: ${DISK_USAGE}% (严重告警!)"
        return 1
    fi

    return 0
}

check_docker_resources() {
    echo "🔍 检查容器资源使用..."

    docker stats --no-stream "${NGINX_CONTAINER}" "${PHP_CONTAINER}" "${DB_CONTAINER}"

    return 0
}

check_network() {
    echo "🔍 检查容器网络..."

    # 检查PHP到DB的连通性
    if docker exec "${PHP_CONTAINER}" ping -c 1 db >/dev/null 2>&1; then
        echo -e "${GREEN}✅${NC} PHP -> DB 网络正常"
    else
        echo -e "${RED}❌${NC} PHP -> DB 网络异常!"
        return 1
    fi

    # 检查Nginx到PHP的连通性
    if docker exec "${NGINX_CONTAINER}" ping -c 1 php >/dev/null 2>&1; then
        echo -e "${GREEN}✅${NC} Nginx -> PHP 网络正常"
    else
        echo -e "${RED}❌${NC} Nginx -> PHP 网络异常!"
        return 1
    fi

    return 0
}

# ========== 主逻辑 ==========
echo "╔════════════════════════════════════════════════════════════╗"
echo "║  🏥 LNMP 健康检查 - by 哈雷酱 (￣▽￣)／                ║"
echo "╚════════════════════════════════════════════════════════════╝"
echo ""
echo "检查时间: $(date '+%Y-%m-%d %H:%M:%S')"
echo ""

# 执行所有检查
FAILED=0

check_nginx || FAILED=$((FAILED + 1))
echo ""

check_php || FAILED=$((FAILED + 1))
echo ""

check_database || FAILED=$((FAILED + 1))
echo ""

check_network || FAILED=$((FAILED + 1))
echo ""

check_disk_space || FAILED=$((FAILED + 1))
echo ""

check_docker_resources
echo ""

# ========== 结果汇总 ==========
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
if [ ${FAILED} -eq 0 ]; then
    echo -e "${GREEN}🎉 所有检查通过! 系统运行正常!${NC}"
    exit 0
else
    echo -e "${RED}❌ 检查失败: ${FAILED} 项异常!${NC}"
    echo "请查看上面的错误信息并及时处理!"
    exit 1
fi
```

### 使用方法:
```bash

# 添加:

```

## 3. 日志清理脚本

### cleanup-logs.sh - 智能日志清理

```bash
#!/bin/bash
# ============================================================
# 日志清理脚本
# 功能: 定期清理和归档日志文件
# ============================================================

PROJECT_DIR="/home/monika/my-LNMP-project"
LOGS_DIR="${PROJECT_DIR}/logs"
ARCHIVE_DIR="${PROJECT_DIR}/logs/archives"
RETENTION_DAYS=30

echo "🗑️  开始日志清理..."

# 创建归档目录
mkdir -p "${ARCHIVE_DIR}"

# 归档旧日志
TIMESTAMP=$(date +%Y%m%d-%H%M%S)
ARCHIVE_FILE="${ARCHIVE_DIR}/logs-${TIMESTAMP}.tar.gz"

echo "📦 归档日志到: ${ARCHIVE_FILE}"
tar -czf "${ARCHIVE_FILE}" \
    "${LOGS_DIR}/nginx"/*.log \
    "${LOGS_DIR}/php"/*.log \
    2>/dev/null || true

# 清空当前日志
echo "🧹 清空当前日志文件..."
sudo sh -c "> ${LOGS_DIR}/nginx/access.log"
sudo sh -c "> ${LOGS_DIR}/nginx/error.log"
sudo sh -c "> ${LOGS_DIR}/php/error.log" 2>/dev/null || true

# 删除旧归档
echo "🗑️  删除${RETENTION_DAYS}天前的归档..."
find "${ARCHIVE_DIR}" -name "logs-*.tar.gz" -mtime +${RETENTION_DAYS} -delete

echo "✅ 日志清理完成!"
```

## 4. 性能监控脚本

### monitor-performance.sh - 性能实时监控

```bash
#!/bin/bash
# ============================================================
# 性能监控脚本
# 功能: 实时监控容器性能并记录
# ============================================================

LOG_FILE="/home/monika/my-LNMP-project/logs/performance-$(date +%Y%m%d).log"

while true; do
    TIMESTAMP=$(date '+%Y-%m-%d %H:%M:%S')

    # 记录时间戳
    echo "========== ${TIMESTAMP} ==========" >> "${LOG_FILE}"

    # 记录容器资源
    docker stats --no-stream my_lnmp_nginx my_lnmp_php my_lnmp_db >> "${LOG_FILE}"

    # 记录磁盘空间
    echo -e "\n磁盘使用:" >> "${LOG_FILE}"
    df -h / >> "${LOG_FILE}"

    echo -e "\n" >> "${LOG_FILE}"

    # 每5分钟记录一次
    sleep 300
done
```

## 5. 自动部署脚本

### deploy.sh - 零宕机部署

```bash
#!/bin/bash
# ============================================================
# 自动部署脚本
# 功能: 零宕机滚动更新部署
# ============================================================

set -e

echo "🚀 开始部署流程..."

# 1. 备份数据库
echo "📦 Step 1: 备份数据库..."
./backup-database.sh

# 2. 拉取最新代码(如果是git仓库)
if [ -d ".git" ]; then
    echo "📥 Step 2: 拉取最新代码..."
    git pull origin main
fi

# 3. 构建新镜像
echo "🔨 Step 3: 构建新镜像..."
docker compose build --no-cache php

# 4. 滚动更新PHP服务
echo "🔄 Step 4: 更新PHP服务..."
docker compose up -d --no-deps --build php

# 5. 等待服务稳定
echo "⏳ Step 5: 等待服务稳定..."
sleep 10

# 6. 健康检查
echo "🏥 Step 6: 健康检查..."
if ./health-check.sh; then
    echo "✅ 部署成功!"
    exit 0
else
    echo "❌ 部署失败! 请检查日志!"
    exit 1
fi
```

## 6. 故障恢复脚本

### disaster-recovery.sh - 灾难恢复

```bash
#!/bin/bash
# ============================================================
# 灾难恢复脚本
# 功能: 快速恢复服务到最近的备份点
# ============================================================

set -e

BACKUP_DIR="/home/monika/my-LNMP-project/backups"
DB_CONTAINER="my_lnmp_db"
DB_ROOT_PASSWORD="liu20041016"

echo "⚠️  灾难恢复模式"
echo ""

# 1. 显示可用备份
echo "📋 可用的备份文件:"
ls -lht "${BACKUP_DIR}" | grep "backup-" | head -n 10

echo ""
read -p "请输入要恢复的备份文件名: " BACKUP_FILE

if [ ! -f "${BACKUP_DIR}/${BACKUP_FILE}" ]; then
    echo "❌ 备份文件不存在!"
    exit 1
fi

# 2. 确认恢复
echo ""
echo "⚠️  警告: 这将覆盖所有现有数据!"
read -p "确认恢复? (yes/no): " CONFIRM

if [ "${CONFIRM}" != "yes" ]; then
    echo "已取消"
    exit 0
fi

# 3. 停止服务
echo "⏸️  停止服务..."
docker compose stop nginx php

# 4. 恢复数据库
echo "📥 恢复数据库..."
gunzip < "${BACKUP_DIR}/${BACKUP_FILE}" | \
    docker exec -i "${DB_CONTAINER}" mysql -uroot -p"${DB_ROOT_PASSWORD}"

# 5. 重启服务
echo "🔄 重启服务..."
docker compose start nginx php

# 6. 健康检查
echo "🏥 健康检查..."
sleep 5
./health-check.sh

echo "✅ 恢复完成!"
```
